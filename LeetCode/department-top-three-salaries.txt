WITH RankedEmployees AS (
    SELECT
        d.name AS Department,
        e.name AS Employee,
        e.salary AS Salary,
        DENSE_RANK() OVER (
            PARTITION BY e.DepartmentId
            ORDER BY e.salary DESC
        ) AS dr
    FROM Employee e
    JOIN Department d
        ON e.DepartmentId = d.id
)

SELECT
    Department,
    Employee,
    Salary
FROM RankedEmployees
WHERE dr <= 3;
